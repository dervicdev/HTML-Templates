03 // Tech Standard

Zero-Database Flat-File Core Engine

No MySQL database setup. No SQL injection vulnerabilities. Just ultra-fast JSON storage and raw PHP speed.

core/flatfile_database.php No-SQL Engine 
<?php
declare(strict_types=1);

namespace CodeVault\\Engine;

final class FlatFileDatabase {
    private string $storagePath;

    public function __construct(string $storagePath = '../data/') {
        $this->storagePath = $storagePath;
    }

    public function fetchCollection(string $collection): array {
        $file = $this->storagePath . $collection . '.json';
        if (!file_exists($file)) return [];
        
        $content = file_get_contents($file);
        return json_decode($content, true) ?? [];
    }

    public function insertRecord(string $collection, array $payload): bool {
        $data = $this->fetchCollection($collection);
        $payload['id'] = uniqid('cv_', true);
        $payload['created_at'] = time();
        
        $data[] = $payload;
        return (bool)file_put_contents(
            $this->storagePath . $collection . '.json', 
            json_encode($data, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES)
        );
    }
}
